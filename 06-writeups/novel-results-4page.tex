\begin{abstract}
A \emph{tournament} is a complete directed graph.  Let $H(T)$ count directed
Hamiltonian paths.  We report three previously unknown rigidities.
\textbf{Arithmetic}: the odd integers $7$ and $21$ are permanently absent from
$\{H(T)\}$ over all tournaments on all $n$; and a signed Hamiltonian permanent
$S(T)$ satisfies $S(T)\equiv c_n\pmod{2^{n-1}}$ for a constant $c_n$
independent of $T$.
\textbf{Spectral}: $H(T)$ has a complete closed-form Walsh--Fourier
decomposition; the nonzero coefficients are supported exactly on
unions of edge-disjoint even-length paths in $K_n$, and this
simultaneously yields an elementary proof of the Odd-Cycle Collection
Formula $H(T)=I(\Omega(T),2)$ bypassing $P$-partition theory, and a new link
between the statistics of random tournaments, the Cayley transform, and
Delannoy lattice paths.
\textbf{Topological}: in GLMY path homology, $\beta_2(T)=0$ for every
tournament --- combinatorial completeness forbids the twin-vertex configurations
that force $\beta_2>0$ in general directed graphs --- while Paley tournaments
exhibit a Betti profile matching the cohomology of a Heisenberg Lie algebra.
\end{abstract}

\section{Background and the Odd-Cycle Collection Formula}

A \textbf{tournament} $T$ on $[n]=\{1,\dots,n\}$ is a complete directed graph;
$H(T)$ denotes the number of directed Hamiltonian paths.  By R\'edei's theorem
(1934), $H(T)$ is always odd.
The \textbf{conflict graph} $\Omega(T)$ has one vertex per directed odd cycle of
$T$, with edges between cycles sharing a vertex.

\begin{thm}[Odd-Cycle Collection Formula (OCF)]\label{thm:ocf}
For every tournament $T$,\quad $H(T)=I(\Omega(T),2)$,
where $I(G,x)=\sum_{k\ge 0}\alpha_k(G)x^k$ is the independence polynomial.
\end{thm}

The formula was discovered computationally here, verified exhaustively for
$n\le 6$ (33,864 labeled tournaments), and sampled for $n=7$--$10$ with zero
failures.  An independent proof using $P$-partitions appears in \cite{GS24}.
Section~\ref{sec:walsh} gives a new, elementary Walsh--Fourier proof.
The expansion $H=1+2\alpha_1+4\alpha_2+\cdots$
recovers R\'edei's theorem and identifies $H=1$ with transitivity.

\section{Arithmetic Rigidity}\label{sec:arith}

\subsection*{The $H$-spectrum gap problem}

Which odd integers are \emph{permanently forbidden} --- absent from $\{H(T)\}$
over all tournaments on all $n$?

\begin{thm}[Gap Theorem]\label{thm:gap}
$H(T)\ne 7$ and $H(T)\ne 21$ for every tournament $T$, regardless of $n$.
These are the only permanently forbidden values below $200$; $H=63$ is
achieved at $n=8$.
\end{thm}

\begin{proof}[Proof sketch for $H\ne 7$]
By Theorem~\ref{thm:ocf}, $H=7$ requires $\Omega(T)=K_3$ (exactly three odd
cycles forming a complete conflict graph).  One shows: for $n\le 7$, no
tournament has exactly three pairwise-conflicting cycles (exhaustive at $n\le 7$,
using the fact that at $n=5$ three mutually-conflicting $3$-cycles always
generate a $5$-cycle).  For $n\ge 8$, the three cycles span at most $7$
vertices, so the induced subtournament on those vertices reduces to the $n\le 7$
case.
\end{proof}

The proof of $H\ne 21$ is deeper, proceeding by a case analysis of possible
conflict graph structures realizing $I(G,2)=21$: the disconnected case is
ruled out using $H\ne 7$; the path $P_4$ is ruled out by a forced-cycle
lemma; the remaining cases are closed inductively via a ``poisoning-DAG''
argument showing every sufficiently large cycle-rich tournament either has
a good deletion vertex (reducing to a smaller case) or has $H\ge 25>21$.

\smallskip
\noindent\textbf{Connection to $k$-nacci matrices.}
Let $M_k$ be the companion matrix of the $k$-th order Fibonacci recurrence
$a_n=a_{n-1}+\cdots+a_{n-k}$.  Newton's identity gives
$\Tr(M_k^3)=7$ for \emph{all} $k\ge 3$, since the first three elementary
symmetric polynomials of $k$-nacci roots are identical for all such $k$.
For $k=3$ (tribonacci), $\Tr(M_3^5)=21=3\times 7$.

\begin{thm}[$k$-nacci Identity]\label{thm:mersenne}
$\Tr(M_k^n) = 2^n-1$ for all $1\le n\le k$.
At $n=k+1$ the identity breaks by exactly $k+1$: $\Tr(M_k^{k+1})=2^{k+1}-k-2$.
\end{thm}

The forbidden values $\{7,21\}$ are thus the first two $k$-nacci traces that
lie strictly below their corresponding power-of-two targets.

\subsection*{The signed Hamiltonian permanent}

Let $B=2A-J$ ($\pm 1$ signed adjacency matrix) and
$S(T)=\sum_{P}\prod_{i=1}^{n-1}B[P_i,P_{i+1}]$
over all $n!$ vertex orderings.  One has $S(T)=0$ for even $n$ by a
reversal-pairing argument.

\begin{thm}[Universal Congruences]\label{thm:signed}
\textup{(i)} $S(T)\bmod 2^{n-1}$ depends only on $n$.

\textup{(ii)} $S(T)$ is fully independent of $T$ modulo $2^{n-1}$
if and only if $s_2(n-3)\le 1$, where $s_2$ is the binary digit sum.
The universal odd values of $n$ are $3,5,7,11,19,35,67,\ldots$
\textup{(}i.e., $n=3+2^k$ for $k\ge 0$, together with $n=3$\textup{)}.

\textup{(iii)} \emph{(Master Identity.)} For any $k$ pairwise non-adjacent
positions in a Hamiltonian path,
$D_S = n!/2^k$
holds pointwise for all tournaments, where $D_S$ sums weighted edge products
over all $(2k)!$ orderings of $2k$ vertices at those positions.
\end{thm}

At the first non-universal value $n=9$, the residue $S(T)\bmod 256$
depends precisely on the parity of the $3$-cycle count $t_3(T)$.

\section{Spectral Rigidity: Walsh--Fourier Theory}\label{sec:walsh}

Encode $T$ as $t\in\{0,1\}^m$, $m=\binom{n}{2}$.  Every tournament invariant
expands in the Walsh--Hadamard basis $\{(-1)^{t\cdot\chi_S}\}_{S\subseteq[m]}$.

\begin{thm}[Walsh Spectrum of $H$]\label{thm:walsh}
$\hat H[S]\ne 0$ if and only if $S$ is a union of edge-disjoint even-length
paths in $K_n$.  For such $S$ with $|S|=2k$ and $r$ path components,
\[
\hat H[S] \;=\; \varepsilon_S\cdot\frac{2^r\cdot(n-2k)!}{2^{n-1}},
\qquad \varepsilon_S=\pm 1\ \text{\textup{(}path-orientation parity\textup{)}}.
\]
\end{thm}

\begin{proof}[Proof]
For a path-union monomial $S$, the inner product $\hat H[S]=2^{-m}\sum_T
H(T)(-1)^{t\cdot\chi_S}$ counts Hamiltonian paths weighted by arc consistency
with $S$.  Each component of $S$ must appear as a contiguous block in any
contributing path; even-length components contribute sign $+1$ regardless of
traversal direction (two orientations, each with sign product $(-1)^{2a}=1$),
giving $2^r(n-2k)!$ oriented placements.  For non-path-union $S$ the sum
cancels by reversal symmetry.  Setting $\hat H[S]=\hat I_\Omega[S]$ via an
independent generating-function calculation proves Theorem~\ref{thm:ocf}.
\end{proof}

The formula compresses $H(T)$ at $n=7$ from $2^{21}$ configurations to
$\approx 20$ independent Walsh amplitudes ($10^5$-fold reduction).

\begin{cor}[Transfer Matrix Symmetry]\label{cor:transfer}
Let $M[a,b]=\#\{\text{directed Hamiltonian paths }a\to b\}$.
Then $M[a,b]=M[b,a]$ for all tournaments and all vertices $a,b$.
\end{cor}

This is non-obvious: the matrix counting directed-path endpoints is symmetric
despite the underlying directed structure.  It follows from an explicit
closed-form Walsh expansion of $M[a,b]$, manifestly symmetric in $a,b$.

\subsection*{The Cayley--Delannoy connection}

Let $g_k(m)$ denote the Taylor coefficients of the Cayley transform:
$Q(x)^m=\bigl(\frac{1+x}{1-x}\bigr)^m=1+2\sum_{k\ge 1}g_k(m)\,x^k$.

\begin{thm}[Delannoy Identity]\label{thm:delannoy}
\[
k\cdot g_k(m) \;=\;
\sum_{j=1}^{\min(k,m)} j\tbinom{k}{j}\tbinom{m}{j}2^{j-1}
\;=\; \#\bigl\{\text{diagonal steps in all Delannoy paths }(0,0)\to(k,m)\bigr\}.
\]
The squared coefficient of variation of $H(T)$ over uniformly random
tournaments satisfies
\[
\CV^2(H)=\sum_{k\ge 1}\frac{2\,g_k(n-2k)}{(n)_{2k}}
=\frac{2}{n}+\frac{0}{n^2}-\frac{14}{3n^3}+O(n^{-4}).
\]
The sequence $W(n)=n!\,(1+\CV^2(H))=1,2,8,32,158,928,6350,49752,439670,
4327904,46963358,\ldots$ (through $n{=}19$ via $O(n^2{2^n})$ bitmask DP)
is not in OEIS.
\end{thm}

The exact vanishing of the $n^{-2}$ coefficient follows from $g_2=g_1^2$.

\begin{rem}
The \emph{$H$-weighting identity} $|\{T: H{=}h\}|_{\text{lab}} = (n!/h)\cdot|\{t: H{=}h\}|_{\text{tile}}$
(CUT$\oplus$CYCLE, verified $n{\le}8$) gives a $2^{n-1}$-fold speedup.
The complete $n{=}8$ H-distribution (first computation; $29\,\mathrm{s}$) confirms:
$H{\ne}7$, $H{\ne}21$; $H_{\max}{=}661$ (prime); all odd values in
$[1,599]\setminus\{7,21\}$ achieved; $\Sigma H(8)=W(8)\cdot 2^{14}$ exactly.
Iso-class counts follow by $|\{I: H{=}h\}|=\mathsf{tile}(h)/h$ when
all $|\mathrm{Aut}|{=}1$ at $H{=}h$: at $n{=}7$, $\mathsf{tile}(189)/189=1/21$
identifies the Paley tournament $T_7$ ($|\mathrm{Aut}|{=}21$, verified).
\end{rem}
\begin{rem}[$q$-deformed tournament count]
$A(n,q)=\!\sum_{\lambda\vdash n,\,\text{odd}}\!q^{c(\lambda)}/z(\lambda)$ satisfies
$A(n,2){=}A_{000568}(n)$ and $A(m,4){=}\mathrm{SC}(2m)$ (proved via
$c(2\lambda){=}2c(\lambda){+}|\lambda|$, $z(2\lambda){=}2^{|\lambda|}z(\lambda)$, so
$2^{c(2\lambda)}/z(2\lambda){=}4^{c(\lambda)}/z(\lambda)$).
Runs in $O(p_\mathrm{dist}(n))$; computes both sequences through $n{=}30$ in ${<}1\,\mathrm{s}$.
\end{rem}

The Cayley transform $Q(x)=(1+x)/(1-x)$ -- the bilinear/Tustin transform
of digital signal processing -- governs the statistics of random tournaments
through weighted Delannoy path enumeration; no prior literature records this.
Its logarithm $2\operatorname{arctanh}(x)$ is simultaneously the Poincar\'e
disk geodesic and the Fisher--Rao distance on Bernoulli distributions:
\emph{the Fourier energy of a random tournament equals the statistical distance
between its arc-distribution and the fair coin.}

\section{Topological Rigidity: GLMY Path Homology}\label{sec:homology}

GLMY path homology \cite{GLMY12} assigns Betti numbers $\beta_p(T)$ via an
``allowed paths'' chain complex.  For general oriented graphs $\beta_2>0$
occurs freely (e.g.\ among $5$-vertex oriented graphs); for tournaments it is
structurally forbidden.

\begin{thm}[$\beta_2=0$]\label{thm:beta2}
For every tournament $T$ on every number of vertices, $\beta_2(T)=0$.
\end{thm}

\begin{proof}[Proof sketch]
Induction via the long exact sequence of the pair $(T,T\setminus v)$.
By induction $H_2(T\setminus v)=0$, so $H_2(T)$ injects into the relative term,
which vanishes iff $\beta_1(T\setminus v)\le\beta_1(T)$; it suffices to find
one such \emph{good vertex} in every tournament.  Four cases cover all
possibilities: if $\beta_1(T)=1$, every vertex is good; if $T$ is not strongly
connected, $\beta_1$ is $0$ throughout and every vertex is good; if $T$ has a
cut vertex, that vertex is good; for $2$-connected $T$ at $n\ge 6$, a Key Lemma
(no cycle can be completely isolated from dominated cycles in a $2$-connected
tournament) produces at least $n-5$ good vertices; and $n\le 5$ is checked
exhaustively.
\end{proof}

\noindent\textbf{Structural explanation.}
Every oriented graph with $\beta_2>0$ contains \emph{twin vertices} (two
vertices with identical neighborhoods).  Tournament completeness forbids twins:
if $u,v$ shared the same in-neighborhood, the arc between $u$ and $v$ would
simultaneously be required in both directions.

\smallskip
Further proved results:
$\beta_1(T)\in\{0,1\}$ (no tournament has more than one independent $1$-cycle);
$\beta_1(T)\cdot\beta_3(T)=0$ (no tournament simultaneously carries $1$-cycles
and $3$-cycles);
$\rank(\partial_2)=\binom{n}{2}-n+1-\beta_1(T)$ universally.
At $n=8$, $\beta_3=2$ first occurs ($0.08\%$ of tournaments) and the
seesaw $\beta_3\cdot\beta_4=0$ fails, opening new structure.

\begin{thm}[Paley Betti Formula; computationally established for $p=7,11$]%
\label{thm:paley}
For the Paley tournament $T_p$ \textup{(}$p\equiv 3\pmod 4$ prime,
$a\to b$ iff $b-a$ is a nonzero quadratic residue mod $p$, $m=(p-1)/2$\textup{)}:
\[
\beta_m = \frac{m(m-3)}{2},\quad
\beta_{m+1}=\tbinom{m+1}{2},\quad
\beta_d=0\ (d\ne 0,m,m+1),\quad
\chi(T_p)=p.
\]
\end{thm}

The formula $\beta_m=m(m-3)/2$ equals exactly the second Betti number of the
$m$-dimensional Heisenberg Lie algebra (Santharoubane \cite{S83}); this
identification appears to be new.
Key computations: $H(T_{11})=95{,}095$ and
$H(T_{11})/|\mathrm{Aut}(T_{11})|=1729$ (Hardy--Ramanujan taxicab number).

\begin{thm}[Constant Symbol Matrix]\label{thm:csm}
For any circulant tournament, the Tang--Yau symbol matrix $M_m(t)$
\textup{\cite{TY26}} is constant in $t$: all $n$ eigenspaces of the
$\mathbb{Z}_n$-action on the chain complex have identical $\Omega$-dimension
vectors.
\end{thm}

\noindent\emph{Proof.}
In Tang--Yau's framework, $t$-powers arise only from face-$0$ maps that
land outside the allowed-path space $A_{m-1}$.  For a circulant
\emph{tournament} (antisymmetric connection set $S\cap(-S)=\varnothing$),
one shows: the tail of any allowed $m$-path is always an allowed
$(m{-}1)$-path (Key Lemma, proved by comparing partial sums modulo $n$).
Hence face-$0$ maps never leave $A_{m-1}$, so $M_m(t)=M_m(1)$ is constant.
\hfill$\square$

\smallskip
In \cite{TY26} the symbol matrix varies with $t$ for general circulant
digraphs; Theorem~\ref{thm:csm} is specific to tournaments and strictly
sharpens their stability theorem.  Combined with small-prime Gaussian
elimination (uint8 mod $p<256$, $8\times$ memory reduction, certified at two
primes), the complete Betti profile of $T_{11}$ was computed; $T_{19}$ is
within reach.

\section{Open Problems}

\begin{enumerate}
\item \textbf{Gap density.} Are $7$ and $21$ the only permanently forbidden
$H$-values?  Confirmed at $n{\le}8$: every odd value in $[1,599]\setminus\{7,21\}$
is achieved; near $H_{\max}(8){=}661$ nine additional values are absent.
Does achievable density approach~$1$?
\item \textbf{Paley Betti formula.} Prove Theorem~\ref{thm:paley} for all Paley
primes; the natural approach uses the Heisenberg isomorphism and the
representation theory of $\mathbb{Z}_p\rtimes\mathbb{Z}_m$.
\item \textbf{Paley maximization.} Prove $H(T_p)=\max_{|V|=p}H(T)$ for all
$p\equiv 3\pmod 4$.  Verified: $p=3,7,11$; matches OEIS A038375.
\item \textbf{$\beta_3=2$ structure.} Characterize the $0.08\%$ of $8$-vertex
tournaments with $\beta_3=2$.  Is $\sup_T\beta_k(T)$ unbounded in $n$?
\item \textbf{Universality mechanism.} Give a direct $2$-adic or representation-theoretic
explanation for the universality criterion $s_2(n-3)\le 1$
in Theorem~\ref{thm:signed}\textup{(ii)}.
\item \textbf{$W(n)$ recursion.} No 2-term recurrence $(an{+}b)W_{n-1}+(cn{+}d)W_{n-2}$
fits the sequence $W(n)=1,2,8,32,158,928,\ldots$ (checked through $n{=}19$).
Find a recursion or a closed form; the ratio $W(n)/n!\to 1$ raises the question
of the rate $\mathrm{CV}^2=W/n!-1\approx 2/n$ and its correction terms.
\end{enumerate}
