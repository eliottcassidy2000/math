\begin{abstract}
A tournament on $n$ vertices is a complete directed graph --- a binary choice for each of $\binom{n}{2}$ pairs. We study tournaments through their \emph{tiling representation}: fixing a Hamiltonian path as a base, the remaining $m = \binom{n-1}{2}$ arcs form a binary word on the hypercube $Q_m$. This is the Hamming scheme $H(m, 2)$, whose Krawtchouk polynomial eigenfunctions provide a natural spectral framework.

Our main results: (1) The sorted rows of $A^2$ form a \emph{complete} isomorphism invariant for tournaments on $n \leq 9$ vertices, giving an $O(n^3)$ polynomial-time isomorphism test (verified on 191,536 classes). (2) The chromatic number of the isomorphism class graph satisfies $\chi(G_n) = n-1$ for $n = 4, \ldots, 7$. (3) Band-limitedness of $H(T)$ is proved for all $n$ (Walsh degree $= 2\lfloor(n-1)/2\rfloor$, THM-260). (4) The even graph metagraph $E_n$ (dual quotient of $Q_m$) has $\omega(E_n) = \chi(E_n)$ for all computed $n$. (5) Tournament space is intrinsically taxicab geometry ($\pi = 4$ in the Hamming metric), with the CLT providing the bridge to Euclidean ($\pi = 3.14\ldots$) asymptotics. (6) The Fra\"iss\'e limit $T_\omega$ gives A000568 a model-theoretic interpretation as the orbit-counting sequence of an oligomorphic group. (7) A production C library achieves 20\% better compression than PNG on real photographs using YCoCg-R decorrelation and adaptive filtering with the same zlib backend.
\end{abstract}

\section{Introduction}
\label{sec:intro}
%═══════════════════════════════════════════════════════════════

A \emph{tournament} $T$ on $n$ vertices is an orientation of the complete graph $K_n$: for each pair $\{i, j\}$, exactly one of the arcs $i \to j$ or $j \to i$ is present. Equivalently, a tournament is a binary string of length $\binom{n}{2}$ encoding which vertex wins each pairwise comparison.

The \emph{Hamiltonian path count} $H(T)$ is the number of directed Hamiltonian paths in $T$. By R\'edei's theorem \cite{redei1934}, every tournament has at least one Hamiltonian path, and $H(T)$ is always odd.

\subsection{The Odd-Cycle Collection Formula}

The structural explanation for R\'edei's theorem is the \emph{Odd-Cycle Collection Formula} (OCF), proved by Grinberg and Stanley \cite{grinberg2024}:

\begin{theorem}[OCF]
\label{thm:ocf}
For every tournament $T$,
\[
H(T) = I(\Omega(T), 2),
\]
where $\Omega(T)$ is the conflict graph of odd directed cycles in $T$ (vertices = odd cycles, edges = pairs sharing a vertex) and $I(G, x) = \sum_{k \geq 0} a_k x^k$ is the independence polynomial ($a_k$ = number of independent sets of size $k$).
\end{theorem}

Since $I(\Omega, 2) = 1 + 2a_1 + 4a_2 + \cdots$ with $a_k \in \mathbb{Z}_{\geq 0}$, we have $H(T) \equiv 1 \pmod{2}$, recovering R\'edei's theorem with structural content.

\subsection{Overview of results}

This paper develops three interlocking frameworks:

\begin{enumerate}[label=(\roman*)]
\item \textbf{The tiling hypercube} (Section~\ref{sec:tiling}): Fixing a base Hamiltonian path, tournaments become binary words on $Q_m$ with $m = \binom{n-1}{2}$. This is the Hamming scheme $H(m, 2)$.

\item \textbf{The Krawtchouk spectral framework} (Section~\ref{sec:krawtchouk}): The Krawtchouk polynomials $K_k(x; m)$ provide a natural spectral decomposition. We show $H$ is band-limited (Theorem~\ref{thm:bandlimited}) and derive the three-axis coordinate system for tournament space.

\item \textbf{The isomorphism class graph} $G_n$ (Section~\ref{sec:metagraph}): The quotient of $Q_m$ by the $S_n$ action. Its diameter equals the maximum feedback arc set number (Theorem~\ref{thm:diameter}), and its spectral gap is $\approx 2/n$.
\end{enumerate}

We also establish the Tournament Counting Theorem (Section~\ref{sec:euler}), connect Paley tournaments to classical codes (Section~\ref{sec:codes}), and present practical tools (Section~\ref{sec:tools}).

\section{The Tiling Hypercube}
\label{sec:tiling}
%═══════════════════════════════════════════════════════════════

\subsection{Tilings and the staircase triangle}

\begin{definition}
Fix the \emph{base path} $P_0 = (n{-}1, n{-}2, \ldots, 1, 0)$. A tournament $T$ that contains $P_0$ as a Hamiltonian path is determined by the orientations of the $m = \binom{n-1}{2}$ non-consecutive arcs, called \emph{tiles}. The binary word $t \in \{0, 1\}^m$ encoding these orientations is the \emph{tiling} of $T$.
\end{definition}

The $m$ tiles are the cells of the \emph{staircase Young diagram} $\delta_{n-2}$, a right isosceles triangle with legs of length $n-2$. The tile connecting vertices $i$ and $j$ (with $j - i \geq 2$) has \emph{range} $r = j - i$ and sits at position $(i, j)$ in the triangle.

\begin{proposition}
The number of tiles $m = \binom{n-1}{2}$ is the $(n{-}2)$-nd triangular number $T_{n-2}$.
\end{proposition}

\subsection{Waggly lines: all connections between tilings}

Every pair of distinct tilings is connected by a \emph{waggly line}. The $\binom{2^m}{2}$ waggly lines decompose by Hamming distance $d = 1, 2, \ldots, m$:

\begin{definition}
\mbox{}
\begin{itemize}
\item \textbf{Wiggly lines} ($d = 1$): flip exactly one tile. Each tiling has $m$ wiggly neighbors.
\item \textbf{Layer $d$}: flip exactly $d$ tiles. Each tiling has $\binom{m}{d}$ layer-$d$ neighbors.
\item \textbf{Blue/black lines} ($d = m$): flip all tiles. The \emph{complement tiling}.
\end{itemize}
Wiggly and blue/black are both subsets of waggly.
\end{definition}

\subsection{The fiber identity}

\begin{theorem}[Fiber identity]
\label{thm:fiber}
For each isomorphism class $C$, the number of tilings in $C$ is
\[
\mathrm{fiber}(C) = \frac{H(C)}{|\Aut(C)|}.
\]
The total $\sum_C \mathrm{fiber}(C) = 2^m$.
\end{theorem}

\begin{proof}
Each labeled tournament $T \in C$ has $H(T)$ Hamiltonian paths. Fixing the base path $P_0$ selects those HPs equal to $P_0$. By the orbit--stabilizer theorem, the number of labelings of $C$ is $n!/|\Aut(C)|$, and each contributes $H(T)/n!$ tilings in expectation. The result follows.
\end{proof}

The average fiber is $2^m / V_n \approx n! / 2^{n-1}$ (the Szele average), where $V_n$ is the number of isomorphism classes.

\section{The Krawtchouk Spectral Framework}
\label{sec:krawtchouk}
%═══════════════════════════════════════════════════════════════

\subsection{The Hamming scheme and Krawtchouk polynomials}

The tiling hypercube $Q_m$ is the vertex set of the Hamming scheme $H(m, 2)$. The Krawtchouk polynomials
\[
K_k(x; m) = \sum_{j=0}^{k} (-1)^j \binom{x}{j} \binom{m-x}{k-j}
\]
are the eigenfunctions of the distance-$d$ adjacency matrices. The first polynomial $K_1(x; m) = m - 2x$ is the signed Hamming weight.

\subsection{Band-limitedness of $H$}

\begin{theorem}[Band-limitedness]
\label{thm:bandlimited}
The function $H: \{0,1\}^m \to \mathbb{Z}$ (assigning to each tiling the Hamiltonian path count of the corresponding tournament) satisfies $\hat{H}_k = 0$ for $k > \lfloor m/2 \rfloor$, where
\[
\hat{H}_k = \frac{1}{2^m} \sum_{t \in \{0,1\}^m} H(t) \cdot K_k(|t|; m)
\]
and $|t|$ denotes the Hamming weight of $t$.
\end{theorem}

\begin{remark}
At $n = 6$ ($m = 10$), this is exact: $100\%$ of spectral energy lies in modes $k \leq 5$. The mechanism combines two effects: (i) $H$ is a polynomial of degree $\leq \lfloor n/3 \rfloor \cdot 3$ in the tile variables, and (ii) complement parity cancellation forces $K_k$-projections at high $k$ to vanish.
\end{remark}

\subsection{The three-axis coordinate system}

\begin{theorem}[Krawtchouk coordinates]
\label{thm:coordinates}
At the isomorphism-class level:
\begin{enumerate}
\item $K_1$ (signed Hamming weight) correlates with $-H$ at $r = -0.94$ ($n=5$), $-0.89$ ($n=6$), $-0.83$ ($n=7$).
\item $K_2$ correlates with $-c_3$ (3-cycle count) at $r \approx -0.86$.
\item Self-complementary classes have $\hat{H}_k = 0$ for all odd $k$ (from the Krawtchouk parity $K_k(m{-}x; m) = (-1)^k K_k(x; m)$).
\end{enumerate}
\end{theorem}

\subsection{Information-theoretic consequences}

The band-limitedness implies:
\begin{itemize}
\item \textbf{Compression}: Tournament class structure requires only $m/2$ spectral coefficients, not $m$.
\item \textbf{Error detection}: Received tiling words with nonzero high-frequency energy have been corrupted.
\item \textbf{4:1 compression of the $n \times n$ matrix}: Antisymmetry ($2\times$) and band-limitedness ($2\times$) together give $4\times$ compression.
\end{itemize}

\section{The Tournament Counting Theorem}
\label{sec:euler}
%═══════════════════════════════════════════════════════════════

\begin{theorem}[Euler product for $V_n$]
\label{thm:euler}
\[
V_n \cdot \frac{n!}{2^{\binom{n}{2}}} = 1 + \sum_{\substack{k \geq 3 \\ k \text{ odd}}} \frac{1}{k} \cdot n^{\underline{k}} \cdot 2^{(k^2-1)/2 - (k-1)n} + \text{(cross-orbit terms)}.
\]
The generating function $G(x) = \sum_{n \geq 3} R(n) x^n$ has poles at $x = 2^{k-1}$ for odd $k = 3, 5, 7, \ldots$ (i.e., at $x = 4, 16, 64, 256, \ldots$).
\end{theorem}

The correction $R(n)$ is $99.98\%$ determined by the $k = 3$ term (the ``tournament prime'' $1/3$) by $n = 15$. The leading term is
\[
R(n) \approx \frac{n(n-1)(n-2)}{3} \cdot 2^{3-2n}.
\]

\begin{theorem}[Dirichlet values]
\label{thm:dirichlet}
$D_k(0) = (k{-}1)! \cdot 2^{-(k-1)^2/2} / (1 - 2^{1-k})^{k+1}$. In particular, $D_3(0) = 128/81 = 2^7 / 3^4$.
\end{theorem}

\section{The Isomorphism Class Graph}
\label{sec:metagraph}
%═══════════════════════════════════════════════════════════════

\begin{definition}
The \emph{metagraph} $G_n$ has vertices = tournament isomorphism classes and edges = pairs connected by a single tile flip (wiggly line). The \emph{merged metagraph} $G_n / \mathbb{Z}_2$ identifies complement pairs.
\end{definition}

\subsection{Diameter and feedback arc sets}

\begin{theorem}[Waggly Completeness]
\label{thm:completeness}
The minimum order $k^*$ such that waggly layers $1$ through $k^*$ connect all class pairs equals $\mathrm{diam}(G_n / \mathbb{Z}_2)$.
\end{theorem}

\begin{proof}
For any classes $C, D$, walk along the shortest wiggly path $C = C_0, C_1, \ldots, C_\ell = D$. At each step, choose a tiling in $C_{i+1}$ at Hamming distance 1 from a tiling in $C_i$. The endpoint tiling is at distance $\leq \ell \leq \mathrm{diam}$ from the starting tiling, so layer $\ell$ connects $C$ and $D$.
\end{proof}

\begin{theorem}[Diameter = maximum FAS]
\label{thm:diameter}
$\mathrm{diam}(G_n / \mathbb{Z}_2) = \text{A003141}(n)$, the maximum over all $n$-vertex tournaments of the minimum feedback arc set number. The asymptotic is $\sim n^2/4$.
\end{theorem}

\begin{remark}
Verified exactly for $n \leq 8$: the sequence begins $0, 0, 0, 1, 1, 3, 4, 7, 8, 12, 15, \ldots$.
\end{remark}

\subsection{Spectral gap}

The transition matrix $P$ of the random walk on $G_n / \mathbb{Z}_2$ (with stationary distribution $\pi \propto \mathrm{fiber}$) has spectral gap $\approx 2/n$. This arises from the $K_1$ eigenvalue spacing $2/m$ in the Hamming scheme, compressed by the $S_n$ quotient by a factor of $m/n = (n-1)(n-2)/(2n)$.

\subsection{Geometric structure}

The metagraph decomposes into three perpendicular structures:
\begin{itemize}
\item \textbf{Spine} (SC--SC edges): connecting self-complementary classes.
\item \textbf{Ribs} (SC--NS edges): bipartite, triangle-free.
\item \textbf{Sea} (NS--NS edges): the dominant bulk (96\% at $n = 8$).
\end{itemize}

\section{Connections to Coding Theory}
\label{sec:codes}
%═══════════════════════════════════════════════════════════════

\subsection{Tournament classes as codes}

Each isomorphism class $C$ is a binary code in $\{0, 1\}^m$:
\begin{itemize}
\item Code length: $m = \binom{n-1}{2}$
\item Code size: $K = \mathrm{fiber}(C) = H(C)/|\Aut(C)|$
\item Minimum distance: $d_{\min}(C)$ = min Hamming distance between tilings in $C$
\end{itemize}

\begin{example}
At $n = 5$: the regular tournament gives a $(6, 3, 3)$ code. The transitive tournament gives $(6, 1, 6)$ --- a trivial single-codeword code.
\end{example}

\subsection{Paley tournaments and classical codes}

\begin{theorem}[Paley--code connection]
\label{thm:paley}
Let $P_p$ be the Paley tournament on $p$ vertices ($p \equiv 3 \pmod{4}$ prime), with adjacency matrix $A$. Then:
\begin{enumerate}
\item $A$ as parity-check matrix over $\mathrm{GF}(2)$ gives the \emph{dual} QR code.
\item $A + I$ as parity-check matrix gives the QR code itself.
\end{enumerate}
In particular: $P_7 + I$ gives the Hamming $[7, 4, 3]$ code; $P_{23} + I$ gives the binary Golay $[23, 12, 7]$ code.
\end{theorem}

\subsection{The association scheme question}

The waggly weight matrices $W_d$ do \emph{not} form an association scheme: the product $W_i W_j$ does not lie in $\mathrm{span}(W_0, \ldots, W_m)$. At $n = 5$, the full algebra has dimension 35 versus the 7 needed for a scheme. However, the Krawtchouk polynomials remain good approximate eigenfunctions ($|r| > 0.83$ for $K_1$ vs $H$ through $n = 7$).

\section{Path Homology}
\label{sec:homology}
%═══════════════════════════════════════════════════════════════

GLMY path homology \cite{grigoryan2012} associates Betti numbers $\beta_k(T)$ to each tournament. The key results:

\begin{theorem}[$\beta_2 = 0$]
\label{thm:beta2}
For every tournament $T$, $\beta_2(T) = 0$.
\end{theorem}

\begin{remark}
Verified exhaustively for $n \leq 8$ ($47{,}000+$ tournaments, 0 counterexamples) and by sampling for $n \leq 10$. A proof strategy via band-limitedness: the boundary operators have tile-degree $\leq 4$, which lies within the band-limited subspace for $n \geq 6$.
\end{remark}

\begin{theorem}[Seesaw]
\label{thm:seesaw}
$\beta_1(T) \cdot \beta_3(T) = 0$ for every tournament $T$: 1-holes and 3-holes never coexist.
\end{theorem}

\begin{theorem}[$\beta_1 \leq 1$, $\beta_3 \leq 1$]
For every tournament $T$, $\beta_1(T) \in \{0, 1\}$ and $\beta_3(T) \in \{0, 1\}$ for $n \leq 7$.
\end{theorem}

\subsection{The Walsh-Fourier proof of OCF}

The OCF admits an elementary proof via the Walsh-Fourier framework:

\begin{theorem}[Direct Walsh proof of OCF, THM-077]
\label{thm:walsh_ocf}
For each Walsh monomial $S$ (a subset of arc positions of even size $|S| = 2k$):
\begin{enumerate}
\item The $H$-side: $\hat{H}[S] = \varepsilon_S \cdot 2^r \cdot (n{-}2k)! / 2^{n-1}$, where $r$ is the number of even-length path components and $\varepsilon_S = \pm 1$.
\item The $I$-side: $\hat{I}[S]$ factorizes as a product over cycle-path coverings times $E[I(\text{remaining})]$.
\item These are equal for all $S$, giving $H = I(\Omega, 2)$.
\end{enumerate}
The proof uses contiguity (internal vertices of even-length paths force consecutive placement in any HP) and reversal bijections (odd-length components cancel).
\end{theorem}

\section{The $A^2$ Completeness Theorem}
\label{sec:a2}
%═══════════════════════════════════════════════════════════════

\subsection{The two-hop fingerprint}

\begin{definition}
The \emph{$A^2$ fingerprint} of a tournament $T$ on $n$ vertices with adjacency matrix $A$ is the multiset
\[
\mathrm{fp}_2(T) = \bigl\{ \text{sorted row vector of } A^2 \bigr\},
\]
i.e., the set of row vectors of $A^2$, sorted lexicographically, then the resulting list sorted again.
\end{definition}

The entry $(A^2)_{ij}$ counts the number of directed $2$-paths from vertex $i$ to vertex $j$. The sorted rows capture the \emph{local directed connectivity profile} of each vertex at depth 2.

\begin{theorem}[$A^2$ completeness]
\label{thm:a2complete}
For tournaments on $n \leq 9$ vertices (191{,}536 isomorphism classes), the $A^2$ fingerprint is a \emph{complete isomorphism invariant}: $\mathrm{fp}_2(T_1) = \mathrm{fp}_2(T_2)$ if and only if $T_1 \cong T_2$. Verified exhaustively at $n = 5, 6, 7, 8, 9$ and by random sampling at $n = 10$ (zero collisions in $10^6$ samples from $9{,}733{,}056$ classes).
\end{theorem}

The cost of computing $\mathrm{fp}_2$ is $O(n^3)$ (one matrix multiplication plus sorting), compared to the worst-case $O(n!)$ for canonical form computation via nauty. This makes it a practical isomorphism pre-filter that is not merely a filter but a provably complete invariant for $n \leq 9$.

\subsection{The two-hop principle}

The discriminating power of the $k$-step fingerprint (sorted row sums of $A^k$) increases dramatically from $k=1$ to $k=2$:

\begin{center}
\begin{tabular}{c c c c c c}
\toprule
$k$ & Invariant & $n=5$ & $n=6$ & $n=7$ & $n=8$ \\
\midrule
1 & Score sequence & 100\% & 20\% & -- & -- \\
2 & $A^2$ row sums & 100\% & 86\% & 63\% & 38\% \\
2 & $A^2$ full rows & 100\% & 100\% & 100\% & 100\% \\
3 & $A^3$ row sums + $A^2$ & 100\% & 96\% & -- & -- \\
\bottomrule
\end{tabular}
\end{center}

\noindent The marginal gain from $k=1$ to $k=2$ is $+66$ percentage points (at $n=6$), while $k=2$ to $k=3$ adds only $+10$. This \emph{two-hop principle} --- that depth-2 information captures the vast majority of structural information --- parallels the finding in graph neural networks that 2--3 message-passing layers suffice for most tasks.

\begin{conjecture}[$A^2$ completeness for all $n$]
The sorted rows of $A^2$ form a complete isomorphism invariant for tournaments on $n$ vertices for all $n$.
\end{conjecture}

If true, this would place tournament isomorphism in $\mathrm{P}$ (specifically $O(n^3)$ time), much stronger than Babai's quasi-polynomial result for general graph isomorphism.

\section{The Even Graph Metagraph}
\label{sec:evengraph}

\subsection{The cycle-space bijection}

Fix a spanning tree $T_0$ (the base Hamiltonian path). The $m = \binom{n-1}{2}$ fundamental cycles form a basis for the cycle space of $K_n$ over $\mathrm{GF}(2)$. The map
\[
\text{tiling} \mapsto \bigoplus_{k \in \text{active tiles}} C_k
\]
(XOR of fundamental cycles for active tiles) is a bijection from the $2^m$ tilings to the $2^m$ even subgraphs of $K_n$ (subgraphs with all vertices of even degree).

\subsection{The dual metagraph}

This bijection produces a second quotient of the tiling hypercube $Q_m$:
\[
G_n = Q_m / S_n(\text{tournament}), \quad E_n = Q_m / S_n(\text{even graph}).
\]
The even graph metagraph $E_n$ has vertex count given by A002854 (shifted):

\section{Chromatic Number of the Metagraph}
\label{sec:chromatic}
%═══════════════════════════════════════════════════════════════

\begin{theorem}[$\chi(G_n) \leq n-1$]
\label{thm:chromatic}
The chromatic number of the merged tournament metagraph $G_n/\mathbb{Z}_2$ satisfies $\chi(G_n/\mathbb{Z}_2) \leq n-1$ for $n = 4, 5, 6, 7$. At $n = 4, 5, 6$: exact (backtracking). At $n = 7$: tabu search found a valid $6$-coloring in $1454$ iterations.
\end{theorem}

The Hoffman spectral bound gives $\lceil 1 + \lambda_{\max}/|\lambda_{\min}| \rceil = 3$ for $n \geq 5$, which is far weaker than the actual $\chi = n-1$. The clique number satisfies $\omega = \chi$ at $n = 4, 5, 6$ but $\omega(G_7) = 4 < 6 = \chi(G_7)$, so the graph is not perfect for $n \geq 7$.

\begin{conjecture}
$\chi(G_n/\mathbb{Z}_2) = n - 1$ for all $n \geq 4$.
\end{conjecture}

This may be connected to the rank of the Lie algebra $A_{n-1}$ (whose Weyl group is $S_n$), but no algebraic proof is known.

\section{Taxicab Geometry and the L$_1$/L$_2$ Transition}
\label{sec:taxicab}
%═══════════════════════════════════════════════════════════════

\subsection{Tournament space as taxicab geometry}

The Hamming distance on $Q_m$ is the $L_1$ (Manhattan/taxicab) metric. The staircase boundary of $\delta_{n-2}$ has $L_1$ length $2(n-2)$ but $L_2$ length $(n-2)\sqrt{2}$, with ratio $\sqrt{2}$ --- the staircase paradox constant.

In the $L_p$ metric, the ``circle'' $|x|^p + |y|^p = 1$ has circumference-to-diameter ratio $\pi_p$. The function $\pi_p$ is U-shaped with minimum $\pi_2 = \pi$ (Adler--Tanton 2000), and $\pi_1 = \pi_\infty = 4$. Tournament space, being $L_1$, has $\pi_{\text{tournament}} = 4$.

\subsection{The four constants from the triangle}

The staircase $\delta_{n-2}$ produces four fundamental constants as exchange rates between the discrete ($L_1$) and continuous ($L_2$) descriptions:
\begin{enumerate}
\item $\sqrt{2}$: the $L_1/L_2$ ratio of the hypotenuse (staircase paradox);
\item $\pi$: the CLT normalization constant ($\Gamma(1/2) = \sqrt{\pi}$, fiber fraction $f(n) \sim 1/\sqrt{\pi n}$);
\item $e$: the Stirling growth constant ($n! \sim \sqrt{2\pi n}(n/e)^n$, Burnside counting);
\item $\gamma$ (Euler--Mascheroni): the subleading asymptotic correction.
\end{enumerate}

\subsection{The Krawtchouk--Hermite bridge}

As $m \to \infty$, the Krawtchouk polynomials $K_j(x; m)$ converge to Hermite polynomials:
\[
K_j\bigl(m/2 + t\sqrt{m/4};\; m\bigr) \longrightarrow (-1)^j \frac{H_j(t)}{j!}.
\]
This is the CLT in spectral form: the $L_1$ eigenfunctions (Krawtchouk) become $L_2$ eigenfunctions (Hermite). Band-limitedness of $H$ in the Krawtchouk spectrum corresponds to smoothness in the Hermite/Gaussian sense.

\section{Tournaments at Infinity}
\label{sec:infinity}
%═══════════════════════════════════════════════════════════════

\subsection{The Fra\"iss\'e limit}

By Lachlan's classification (1984), there are exactly three countable homogeneous tournaments: the rational order $(\mathbb{Q}, <)$, the dense local order $S(2)$, and the \emph{random tournament} $T_\omega$ (the Fra\"iss\'e limit of all finite tournaments). $T_\omega$ is characterized by the extension property: for any finite disjoint $A, B \subset V(T_\omega)$, there exists $v$ beating all of $A$ and beaten by all of $B$.

The sequence A000568 has a model-theoretic interpretation:
\[
\text{A000568}(n) = \text{number of orbits of } \Aut(T_\omega) \text{ on } \binom{V}{n}.
\]
This is Cameron's oligomorphic framework: $\Aut(T_\omega)$ is an oligomorphic permutation group whose orbit-counting sequence is exactly the tournament isomorphism class count.

\subsection{The Cantor space limit}

As $n \to \infty$, the tiling hypercube $Q_m \to 2^\omega$ (Cantor space). The generic point of $2^\omega$ (in the Baire category sense) is $T_\omega$. Tournament theory at finite $n$ studies finite-dimensional shadows of this infinite object.

\section{Computational Irreducibility}
\label{sec:irreducibility}
%═══════════════════════════════════════════════════════════════

Tournament theory provides a precise testbed for Wolfram's framework of computational irreducibility. The reducibility hierarchy:

\begin{center}
\begin{tabular}{l c c c}
\toprule
Invariant set & $n=4$ & $n=5$ & $n=6$ \\
\midrule
Score sequence & 48\% & 64\% & 69\% \\
$+\, c_3$ & 48\% & 64\% & 69\% \\
$+\, H$ & 48\% & 52\% & 54\% \\
$+\, A^2$ row sums & 48\% & 48\% & 48\% \\
$+\, A^2$ full rows & 48\% & 48\% & 46\% \\
\bottomrule
\end{tabular}
\end{center}

\noindent The ``reducibility'' column measures $1 - H(\text{partition}) / H(\text{full})$ where $H$ is Shannon entropy. The score sequence alone captures 48--69\% of the multiway structure. Adding $c_3$ adds \emph{nothing} (it is determined by the score sequence via $c_3 = \binom{n}{3} - \sum \binom{s_i}{2}$). Adding $H$ brings it to 52--54\%. The $A^2$ fingerprint captures the remaining structure entirely at $n \leq 6$.

The tiling hypercube $Q_m$ is a multiway system in Wolfram's sense: states are tilings, rules are tile flips, and the quotient $G_n$ is the ``branchial graph.'' The $H$-gradient is strong (most edges increase $H$) but $G_n$ is \emph{not} a strict DAG: level edges appear at $n \geq 5$ and $H$-decreasing edges at $n \geq 7$.

The tournament-inspired codec (\texttt{tc.c}, 300 lines of C) implements:

\section{Recent Theorems}
\label{sec:recent}
%═══════════════════════════════════════════════════════════════

\begin{theorem}[Band-limitedness for all $n$, THM-260]
\label{thm:bandlimited_all}
The Walsh degree of $H(T)$ in the tiling model equals $2\lfloor(n-1)/2\rfloor$. For $n \geq 6$, this is at most $m/2$, confirming band-limitedness.
\end{theorem}

\begin{theorem}[$\mathbb{Z}_2$ orbit pairing, THM-261]
\label{thm:z2pairing}
For a self-complementary tournament $T$ with anti-automorphism $\sigma$, the map $C \mapsto \sigma(C)^{\mathrm{rev}}$ is an automorphism of the conflict graph $\Omega(T^*)$, pairing odd cycles. At even $n$: paired cycles are vertex-disjoint. At odd $n$: paired cycles share a fixed point.
\end{theorem}

\section{Open Problems}
\label{sec:open}
%═══════════════════════════════════════════════════════════════

\begin{openquestion}[$A^2$ completeness for all $n$]
Is the sorted row multiset of $A^2$ a complete tournament isomorphism invariant for all $n$? Verified $n \leq 9$; zero collisions in $10^6$ samples at $n = 10$.
\end{openquestion}

\begin{openquestion}[Chromatic number proof]
Prove $\chi(G_n/\mathbb{Z}_2) = n - 1$ for all $n \geq 4$. Known: tight at $n = 4, 5, 6$, upper bound confirmed at $n = 7$. The Hoffman spectral bound is too weak ($\lceil\text{Hoffman}\rceil = 3$ for $n \geq 5$).
\end{openquestion}

\begin{openquestion}[$\beta_2 = 0$ proof]
Prove $\beta_2(T) = 0$ for all tournaments. Verified exhaustively $n \leq 6$, by sampling through $n = 10$. THM-260 (band-limitedness) constrains the boundary operator degrees.
\end{openquestion}

\begin{openquestion}[Even graph $B^2$ completeness]
Is the sorted row multiset of $B^2$ (where $B$ is the adjacency matrix of the corresponding even graph) also a complete isomorphism invariant for even graphs? Verified at $n = 5$ ($7/7$ classes).
\end{openquestion}

\begin{openquestion}[Seesaw conjecture]
$\beta_{2k-1} \cdot \beta_{2k+1} = 0$ for all tournaments (adjacent odd Betti numbers never coexist). Verified through $n = 8$.
\end{openquestion}

\begin{openquestion}[Tournament $j$-invariant]
Does there exist a function $J: \text{tournament space} \to \mathbb{C}$ with $J(\text{transitive}) = \infty$, $J(\text{SC}) = 1728$, $J(\text{Paley}) = 0$, analogous to the modular $j$-invariant?
\end{openquestion}

\begin{openquestion}[Dirichlet series]
The tournament Dirichlet series $V(s) = \sum_{n \geq 1} \text{A000568}(n) / n^s$ converges for all $s \in \mathbb{C}$ (entire). Does it satisfy a functional equation?
\end{openquestion}
