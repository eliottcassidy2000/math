\begin{problem}[\textbf{Background} --- 10 points]\label{prob:background}

\begin{definition}
A \textbf{$k$-uniform hypergraph} on $n$ vertices is a collection of $k$-element subsets (called \emph{hyperedges}) of $\{1, 2, \ldots, n\}$. Two such hypergraphs are \textbf{isomorphic} if one can be obtained from the other by relabeling vertices.
\end{definition}

\begin{enumerate}[label=(\alph*)]
  \item How many labeled $k$-uniform hypergraphs are there on $n$ vertices?

  \item Compute $\binom{n}{k}$ for $(n, k) = (6, 3)$, $(7, 3)$, $(8, 4)$, $(10, 5)$. Notice how $\binom{n}{k}$ grows with $k$: this is why enumeration gets harder for larger uniformity.

  \item A $k$-uniform hypergraph is \textbf{connected} if, for every partition of the vertex set into nonempty parts $A \cup B$, some hyperedge intersects both $A$ and $B$. Verify that for $k = 2$ this reduces to graph connectivity.
\end{enumerate}
\end{problem}

%% PART 2: BURNSIDE FOR K-UNIFORM
%% ============================================================

\begin{problem}[\textbf{Burnside for $k$-Uniform Hypergraphs} --- 30 points]\label{prob:burnside}

By Burnside's lemma:
\[
H_k(n) \;=\; \frac{1}{n!} \sum_{\sigma \in S_n} 2^{f_k(\sigma)}
\]
where $f_k(\sigma)$ is the number of orbits of $\sigma$ on $\binom{n}{k}$ possible hyperedges.

Grouping by cycle type $\lambda = (r_1^{m_1}, r_2^{m_2}, \ldots)$:
\[
H_k(n) \;=\; \sum_{\lambda \vdash n} \frac{2^{c_k(\lambda)}}{\prod_i r_i^{m_i} \cdot m_i!}
\]

\begin{enumerate}[label=(\alph*)]
  \item \textbf{The $k = 2$ case.} For $k = 2$, the orbit count has a closed form:
  \[
  c_2(\lambda) = \sum_i m_i \left\lfloor \frac{r_i}{2} \right\rfloor + \sum_i \binom{m_i}{2} r_i + \sum_{i < j} m_i m_j \gcd(r_i, r_j).
  \]
  Verify that this gives $H_2(4) = 11$.

  \item \textbf{The $k \geq 3$ case.} For $k \geq 3$, no simple closed-form exists for $c_k(\lambda)$. Instead, use the generating function method:
  \[
  c_k(\lambda) = \frac{1}{L} \sum_{d=0}^{L-1} [x^k]\; \prod_i \bigl(1 + x^{r_i / \gcd(d, r_i)}\bigr)^{m_i \cdot \gcd(d, r_i)}
  \]
  where $L = \mathrm{lcm}(r_1, r_2, \ldots)$. Implement this and verify:

  \begin{center}
  \begin{tabular}{c|rrrrr}
  \toprule
  $n \backslash k$ & 2 & 3 & 4 & 5 \\
  \midrule
  4 & 11 & 2 & 2 & --- \\
  5 & 34 & 34 & 6 & 2 \\
  6 & 156 & 2{,}136 & 156 & 6 \\
  7 & 1{,}044 & 7{,}013{,}320 & 7{,}013{,}320 & 156 \\
  \bottomrule
  \end{tabular}
  \end{center}

  Notice the symmetry $H_k(n) = H_{n-k}(n)$. Why?

  \item \textbf{The bottleneck.} The generating-function method requires a polynomial product for each partition and each divisor $d$. How does cost scale with $n$ and $k$?
\end{enumerate}
\end{problem}

\begin{problem}[\textbf{The Perturbation Expansion} --- 25 points]\label{prob:perturbation}

\begin{theorem}[Spectral Gap for $k$-Uniform Structures]\label{thm:gap-k}
The orbit count $c_k(\lambda)$ is maximized uniquely at the identity partition $\lambda = (1^n)$, with $c_k(1^n) = \binom{n}{k}$. The spectral gap to the next partition satisfies:
\[
\Delta_k \;=\; \binom{n}{k} - c_k(2, 1^{n-2}) \;=\; \binom{n-2}{k-1}
\]
achieved by the single transposition.
\end{theorem}

\begin{enumerate}[label=(\alph*)]
  \item \textbf{Verify.} For $k = 2$: $\Delta_2 = \binom{n-2}{1} = n - 2$. For $k = 3$: $\Delta_3 = \binom{n-2}{2} = \frac{(n-2)(n-3)}{2}$. Confirm these numerically at $n = 8$.

  \item \textbf{The implication.} Higher uniformity $k$ gives \emph{faster} convergence of the perturbation expansion, because $\Delta_k$ grows faster. The coupling constant decays as $g(n) \sim n^k / b^{\binom{n-2}{k-1}}$, which is super-exponential for $k \geq 3$.

  \item \textbf{The identity approximation.} For $k = 3$ at $n = 50$, the identity partition gives $H_3(50) \approx 2^{\binom{50}{3}} / 50!$ as an approximation. How many digits does this have? What is the relative error from the single-transposition correction?

  \item \textbf{Universal formula.} The leading correction for any $k$-uniform Burnside problem at base $b$ is:
  \[
  R_2(n, k, b) \;=\; \binom{n}{2} \cdot b^{-\binom{n-2}{k-1}}.
  \]
  Derive this and verify for $k = 2, 3, 4$ at $n = 10$.
\end{enumerate}

\begin{remark}
The spectral gap $\Delta_k = \binom{n-2}{k-1}$ explains why the OEIS has more terms for smaller $k$: the computation is \emph{harder} when identity dominance is weaker (small $k$), requiring more partitions to be evaluated for exact results. Paradoxically, the sequences with the fewest known terms ($k \geq 4$) are the ones where the perturbation expansion works \emph{best}.
\end{remark}
\end{problem}

\begin{problem}[\textbf{Connected Hypergraphs via Inverse Euler Transform} --- 20 points]\label{prob:euler}

If $a(n)$ counts all unlabeled $k$-uniform hypergraphs and $b(n)$ counts the connected ones:
\[
\sum_{n=0}^{\infty} a(n)\, x^n = \prod_{m=1}^{\infty} \frac{1}{(1 - x^m)^{b(m)}}.
\]

\begin{enumerate}[label=(\alph*)]
  \item \textbf{Implement the inverse.} Given $a(0), a(1), \ldots$, recover $b(n)$ via:
  \begin{align*}
    c(n) &= n \cdot a(n) - \textstyle\sum_{j=1}^{n-1} c(j) \cdot a(n - j), \\
    b(n) &= \frac{1}{n} \textstyle\sum_{d \mid n} \mu(n/d) \cdot c(d).
  \end{align*}
  Verify on A000088 $\to$ A001349 (graphs $\to$ connected graphs).

  \item \textbf{Compute new sequences.} Apply to:
  \begin{itemize}
    \item A000665 $\to$ connected 3-uniform hypergraphs (extend beyond OEIS's 12 terms)
    \item A051240 $\to$ connected 4-uniform (compute at least 20 terms --- no OEIS entry exists)
    \item A051249 $\to$ connected 5-uniform (compute at least 15 terms)
    \item A309860 $\to$ connected 6-uniform (compute at least 10 terms)
  \end{itemize}

  \item \textbf{Submit to OEIS.} Format one new sequence as a b-file and draft an OEIS entry.
\end{enumerate}
\end{problem}

\begin{challenge}[\textbf{The Unified Framework} --- Extra Credit]\label{chal:unified}
\begin{enumerate}[label=(\alph*)]
  \item \textbf{The $k$-uniform triangle.} Extend \href{https://oeis.org/A309865}{A309865} to row 26 or beyond.

  \item \textbf{Closed-form edge formulas.} For $k = 2$, $c_2(\lambda)$ has a closed form via GCDs. For $k = 3$, Howroyd (2018) gave a closed form. Derive or look up $c_3(\lambda)$ and compare speed with the generating-function method. Can you find one for $k = 4$?

  \item \textbf{The connected triangle.} Compute the connected $k$-uniform triangle (an analog of A309865 for connected hypergraphs). This does not exist in the OEIS.

  \item \textbf{The critical base $b_c(n, k)$.} For each $k$, compute the base at which the identity contributes 50\% of the Burnside sum. Show that $b_c(n,k) \to 1$ for all $k$, but faster for larger $k$. This means higher-uniformity counting is ``easier'' from the perturbation perspective, despite being computationally harder per term.

  \item \textbf{The grand picture.} The Burnside perturbation expansion reveals that all these counting problems share a universal structure: the identity dominates, corrections decay exponentially, and the convergence rate is controlled by the spectral gap $\Delta_k = \binom{n-2}{k-1}$. Write a one-page summary of this framework suitable for the introduction of a paper.
\end{enumerate}

\section*{Appendix A: Why These Sequences Are Missing}

\begin{enumerate}
  \item \textbf{No closed-form edge formula for $k \geq 3$.} For graphs, $c_2(\lambda)$ is a polynomial in cycle lengths and GCDs. For $k \geq 3$, a generating-function computation is needed per partition per divisor.

  \item \textbf{Exponential growth in $\binom{n}{k}$.} At $n = 10$, $k = 3$: $2^{\binom{10}{3}} = 2^{120}$ labeled hypergraphs.

  \item \textbf{No unified software.} Major CAS systems support graph enumeration but not general $k$-uniform enumeration with optimization.
\end{enumerate}

\section*{Appendix B: The Inverse Euler Transform}

\textbf{Analogy:} Every integer factors uniquely into primes. Every hypergraph decomposes uniquely into connected components. The Euler transform is ``multiplication''; the inverse is ``factorization.''

\textbf{Forward:} $A(x) = \prod_{m \geq 1} (1 - x^m)^{-b(m)}$.

\textbf{Inverse:}
\begin{align*}
c(n) &= n \cdot a(n) - \textstyle\sum_{j=1}^{n-1} c(j) \cdot a(n-j), \\
b(n) &= \frac{1}{n} \textstyle\sum_{d \mid n} \mu(n/d)\, c(d).
\end{align*}
