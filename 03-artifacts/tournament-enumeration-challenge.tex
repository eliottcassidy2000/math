\section*{Background}

\begin{definition}
A \textbf{tournament} on $n$ vertices is a directed complete graph: for every pair of distinct vertices $\{u, v\}$, exactly one of the arcs $u \to v$ or $v \to u$ is present. Two tournaments are \textbf{isomorphic} if one can be obtained from the other by relabeling vertices.
\end{definition}

\noindent The sequence $T(n)$ counting non-isomorphic tournaments on $n$ vertices is OEIS \href{https://oeis.org/A000568}{A000568}:
\[
1,\; 1,\; 1,\; 2,\; 4,\; 12,\; 56,\; 456,\; 6880,\; 191536,\; 9733056,\; \ldots
\]

\subsection*{History}

The exact formula for $T(n)$ was first published by R.~L.~Davis in 1954 and is a standard result in Harary \& Palmer's \emph{Graphical Enumeration} (1973). Implementations in Maple, PARI, Mathematica, and Python exist on the OEIS page.

\medskip
\noindent This challenge has three layers: (1)~implement the formula correctly, (2)~make it fast enough to extend the OEIS, (3)~understand the deep structure that makes computation possible.

\begin{problem}[\textbf{Warm-Up} --- 10 points]\label{prob:warmup}
\begin{enumerate}[label=(\alph*)]
  \item There are $2^{\binom{n}{2}}$ labeled tournaments on $n$ vertices. Prove this.

  \item List all non-isomorphic tournaments on $n = 4$ vertices, with their score sequences. Verify that $T(4) = 4$.

  \item Compute the average number of Hamiltonian paths in a random labeled tournament.

  \textit{Answer:} $\mathbb{E}[H(T)] = n!/2^{n-1}$, by linearity of expectation. There are $n!$ labeled Hamiltonian paths, each present with probability $(1/2)^{n-1}$.
\end{enumerate}
\end{problem}

%% ============================================================
%% PART 2: THE FORMULA
%% ============================================================

\begin{problem}[\textbf{Davis's Formula} --- 25 points]\label{prob:formula}

\begin{theorem}[Davis 1954]\label{thm:davis}
\[
T(n) \;=\; \sum_{\substack{\lambda \vdash n \\ \text{all parts odd}}} \frac{2^{c(\lambda)}}{\displaystyle\prod_i l_i^{m_i} \cdot m_i!}
\]
where $\lambda = (l_1^{m_1}, l_2^{m_2}, \ldots)$, the sum runs over partitions with all parts odd, and the cycle exponent is:
\[
c(\lambda) \;=\; \sum_i m_i \left\lfloor \frac{l_i}{2} \right\rfloor + \sum_i \binom{m_i}{2} l_i + \sum_{i < j} m_i m_j \gcd(l_i, l_j).
\]
\end{theorem}

The ``all parts odd'' restriction comes from Burnside's lemma: permutations with any even-length cycle fix zero tournaments because an even cycle forces contradictory arc orientations. This is equivalent to Alspach's theorem (1970) that tournament automorphism groups have odd order.

\begin{enumerate}[label=(\alph*)]
  \item Implement this formula and verify against the table in Appendix~A.

  \item The formula can be written as a Python one-liner (Chai Wah Wu, OEIS, 2024):
  \begin{lstlisting}
from itertools import product
from math import prod, factorial, gcd
from fractions import Fraction
from sympy.utilities.iterables import partitions

def A000568(n):
    return int(sum(
        Fraction(
            1 << (sum(p[r]*p[s]*gcd(r,s)
                      for r,s in product(p.keys(), repeat=2))
                  - sum(p.values())) >> 1,
            prod(q**p[q] * factorial(p[q]) for q in p)
        )
        for p in partitions(n) if all(q & 1 for q in p)
    ))
  \end{lstlisting}
  Show that Wu's exponent equals $c(\lambda)$ from Theorem~\ref{thm:davis}.

  \item Profile your implementation. Where does it spend time? At what $n$ does it exceed 10 seconds?
\end{enumerate}
\end{problem}

\begin{problem}[\textbf{Making It Fast} --- 30 points]\label{prob:optimize}

\begin{enumerate}[label=(\alph*)]
  \item \textbf{GCD precomputation.} Build a lookup table $G[i][j] = \gcd(i,j)$ for $1 \leq i, j \leq n$. Replace all \texttt{gcd()} calls with table lookups.

  \item \textbf{Bucket accumulation.}
  Create an array \texttt{bucket[$t$]} indexed by the exponent $t = c(\lambda)$. For each partition, add $L / D(\lambda)$ to the appropriate bucket, where $L = \mathrm{lcm}\{D(\lambda)\}$ and $D(\lambda) = \prod l_i^{m_i} m_i!$. After the loop, compute $\sum_t \texttt{bucket}[t] \cdot 2^t$. This separates the expensive big-integer multiplications (one per distinct exponent) from the cheap additions (one per partition).

  \item \textbf{Parallelization.} The partition loop is embarrassingly parallel. Implement a work-queue strategy.

  \item \textbf{Language choice.} Compare Python vs.\ C/Rust with GMP for the big-integer arithmetic.
\end{enumerate}

%% ============================================================
%% PART 4: THE PERTURBATION THEORY
%% ============================================================

\begin{problem}[\textbf{Why the Identity Dominates} --- 25 points]\label{prob:perturbation}

This is the theoretical heart of the challenge.

\begin{theorem}[Spectral Gap]\label{thm:gap}
The cycle exponent $c(\lambda)$ has a unique maximum at the identity partition $\lambda = (1^n)$, where $c(1^n) = \binom{n}{2}$. The \textbf{spectral gap} to the next-largest value is:
\[
\Delta \;=\; \binom{n}{2} - \max_{\lambda \neq (1^n)} c(\lambda) \;=\; 2n - 4
\]
achieved by the single 3-cycle partition $\lambda = (3, 1^{n-3})$.
\end{theorem}

\begin{enumerate}[label=(\alph*)]
  \item \textbf{Verify the theorem.} Compute $c(\lambda)$ for all odd-part partitions of $n = 10$. Confirm that the maximum is $\binom{10}{2} = 45$ at $\lambda = (1^{10})$, and the second-largest is $45 - 16 = 29$ at $\lambda = (3, 1^7)$.

  \item \textbf{The perturbation expansion.} The spectral gap implies that $T(n)$ is dominated by the identity partition:
  \[
  T(n) \;=\; \frac{2^{\binom{n}{2}}}{n!} \times \bigl[1 + R_3(n) + R_{33}(n) + R_5(n) + \cdots\bigr]
  \]
  where the corrections are:
  \begin{align*}
  R_3(n) &= \frac{16\,n(n-1)(n-2)}{3 \cdot 4^n} && \text{(single 3-cycle)} \\
  R_{33}(n) &= 40\binom{n}{6} \cdot 2^{14-4n} && \text{(two 3-cycles)} \\
  R_5(n) &= \frac{n(n{-}1)(n{-}2)(n{-}3)(n{-}4)}{5} \cdot 2^{12-4n} && \text{(single 5-cycle)}
  \end{align*}
  Derive these formulas by computing $c(\lambda)$ and $z_\lambda$ for each partition type and taking the ratio with the identity contribution.

  \item \textbf{Convergence.} Define the \textbf{coupling constant} $g(n) = 1 - \text{(identity fraction)}$, the total relative correction. Compute $g(n)$ for $n = 3, \ldots, 20$. Verify that:
  \begin{center}
  \begin{tabular}{c|ccccc}
  \toprule
  $n$ & 5 & 8 & 10 & 15 & 20 \\
  \midrule
  $g(n)$ & 0.289 & 0.032 & 0.004 & $1.4 \times 10^{-5}$ & $3.3 \times 10^{-8}$ \\
  \bottomrule
  \end{tabular}
  \end{center}
  What is the asymptotic rate of decay of $g(n)$?

  \item \textbf{Practical impact.} At $n = 100$, $T(100)$ has 1333 digits. The identity partition alone gives the first 1279 digits correctly. Adding the single 3-cycle correction gives 1330 digits. With just three corrections, you have 99.99997\% of the answer. Verify this computationally.

  \item \textbf{The punchline.} At $n = 500$, the identity partition gives a $36{,}420$-digit approximation in under 1 millisecond. No partition enumeration is needed. Explain why this works in one paragraph.
\end{enumerate}

\begin{remark}
The perturbation expansion has a physical interpretation: the Burnside sum is a \emph{partition function} at inverse temperature $\beta = \ln 2$. The identity permutation is the ``vacuum state'' (zero energy). Non-identity permutations are ``excitations'' with energy $\Delta c = \binom{n}{2} - c(\lambda)$. The system is \textbf{asymptotically free}: at large $n$, only the vacuum contributes.
\end{remark}
\end{problem}

\begin{challenge}[\textbf{The General Burnside Series} --- Extra Credit]\label{chal:general}

The perturbation expansion is not specific to tournaments. It applies to \emph{any} Burnside counting problem.

\begin{enumerate}[label=(\alph*)]
  \item \textbf{Unified enumerator.} The Burnside formula for structures on $\binom{n}{2}$ edges generalizes to:
  \[
  a(n; b, \mathcal{P}) \;=\; \sum_{\lambda \in \mathcal{P}(n)} \frac{b^{c(\lambda)}}{\prod l_i^{m_i} m_i!}
  \]
  where $b$ is the base (number of colors per edge) and $\mathcal{P}(n)$ restricts the allowed partitions. Implement a unified enumerator that computes:

  \begin{center}
  \begin{tabular}{llcl}
  \toprule
  OEIS & Object & Base & Partition restriction \\
  \midrule
  A000568 & Tournaments & 2 & odd parts only \\
  A000088 & Simple graphs & 2 & all parts \\
  A000273 & Digraphs & 2 & all parts$^*$ \\
  A000595 & Binary relations & 2 & all parts$^*$ \\
  A000666 & Symmetric relations & 2 & all parts$^\dagger$ \\
  \bottomrule
  \end{tabular}
  \end{center}

  \noindent\small{$^*$Digraphs and binary relations use a modified edge formula with coefficient 2 on the gcd sum. $^\dagger$Symmetric relations add 1 to the self-term.}

  \item \textbf{Convergence rate comparison.} The spectral gap depends on the allowed partition types:
  \begin{itemize}
    \item \textbf{All parts} (graphs, digraphs): gap $= n - 2$ (from single transposition $\lambda = (2, 1^{n-2})$). Coupling decays as $g \sim n \cdot b^{-(n-2)}$.
    \item \textbf{Odd parts only} (tournaments): gap $= 2n - 4$ (from single 3-cycle). Coupling decays as $g \sim n^3 \cdot b^{-(2n-4)}$, which is \emph{faster}.
  \end{itemize}
  Verify that at $n = 12$: tournaments have coupling $4.2 \times 10^{-4}$, digraphs $3.1 \times 10^{-5}$, but graphs $3.5 \times 10^{-2}$. Explain the ranking.

  \item \textbf{The phase diagram.} Define the \textbf{critical base} $b_c(n)$ as the value of $b$ where the identity contributes exactly 50\% of the Burnside sum. Show that $b_c(n) \to 1$ as $n \to \infty$. This means: for \emph{any} base $b > 1$, the perturbation theory eventually converges. There is no phase transition, only a crossover.

  \item \textbf{$k$-uniform hypergraphs.} Extend to structures on $\binom{n}{k}$ edges. The spectral gap grows as $\binom{n-2}{k-1}$, which increases with $k$. Higher-uniformity counting converges \emph{faster}. Verify for $k = 3$ (A000665) and $k = 4$ (A051240).

  \item \textbf{Prime vs.\ composite $n$.} At composite $n$, ``resonance partitions'' $(d^{n/d})$ with $d \mid n$ contribute extra correction terms absent at prime $n$. The excess coupling is 1--18\% (largest at $n = 6$). Show that this is a second-order effect: the leading correction $R_3(n)$ does not depend on the factorization of $n$.
\end{enumerate}
\end{challenge}

\section*{Appendix C: The Perturbation Series}

The Burnside sum decomposes as a series in decreasing contribution:
\begin{align*}
n! \cdot T(n) &= \underbrace{2^{\binom{n}{2}}}_{\text{identity}} + \underbrace{\frac{n(n-1)(n-2)}{3}\,2^{1 + \frac{(n-3)(n-2)}{2}}}_{\text{single 3-cycle}} \\
&\quad + \underbrace{40\binom{n}{6}\,2^{5 + \frac{(n-6)(n-7)}{2} + 2(n-6)}}_{\text{two 3-cycles}} + \underbrace{\frac{n!}{5(n{-}5)!}\,2^{2 + \frac{(n-5)(n-4)}{2} + (n-5)}}_{\text{single 5-cycle}} + \cdots
\end{align*}

At $n = 10$, the contributions as fractions of the total are:

\begin{center}
\begin{tabular}{lrr}
\toprule
Partition type & Contribution & Cumulative \\
\midrule
Identity $(1^{10})$ & 99.618\% & 99.618\% \\
Single 3-cycle $(3, 1^7)$ & 0.365\% & 99.983\% \\
Two 3-cycles $(3^2, 1^4)$ & 0.013\% & 99.995\% \\
Single 5-cycle $(5, 1^5)$ & 0.002\% & 99.997\% \\
All remaining & 0.003\% & 100.000\% \\
\bottomrule
\end{tabular}
\end{center}

\noindent The first two partitions capture 99.98\% of the Burnside sum. This convergence improves exponentially with $n$: at $n = 50$, the identity alone gives 99.99999999999999999999999\% of the answer (24 nines).
